# ----------------------------------
# Logging functions
# ----------------------------------
function log-info{
	param([string]$msg)

	if($GLOBAL:EnableLogging){
		(getLogger).Info($msg)
	}
}

function log-error{
	param([string]$msg)

	if($GLOBAL:EnableLogging){
		(getLogger).Error($msg)
	}
}

# ----------------------------------
# Add a Tool to the Tools Array
# ----------------------------------
function Add-Tool {
    param(
        [Parameter(Mandatory=$true)]
        [string]$Name,

        [Parameter(Mandatory=$true)]
        [string]$Manifest,

        [Parameter(Mandatory=$true)]
        [string]$Handler
    )

    # Validate
    if ([string]::IsNullOrWhiteSpace($Name)) {
        throw "Tool name cannot be empty."
    }
    if ([string]::IsNullOrWhiteSpace($Handler)) {
        throw "Tool '$Name' must have a handler."
    }

    # Create tool object
    $toolObj = [PSCustomObject]@{
        Name     = $Name
        Manifest = $Manifest
        Handler  = $Handler
    }

    # Add to global registry
    $GLOBAL:Tools += $toolObj

    log-info "Tool added: $Name"
}

# ----------------------------------
# Check/process passed-in arguments
# ----------------------------------

if ($_args -ne $null) {

	# Provider
	if ($null -ne $_args.Provider) {
		$GLOBAL:AIProvider = $_args.Provider
	}

	# Model
	if ($null -ne $_args.Model) {
		$GLOBAL:AIModel = $_args.Model
	}

	# API Key
	if ($null -ne $_args.APIKey) {
		$GLOBAL:APIKey = $_args.APIKey
	}

	if ($null -ne $_args.AgentGoal) {
		$GLOBAL:AgentGoal = $_args.AgentGoal
	}

	if ($null -ne $_args.MaxIterations) {
		$GLOBAL:MaxIterations = $_args.MaxIterations
	}

	if ($null -ne $_args.EnabledTools) {

		# EnabledTools may be JSON or native objects
		if ($_args.EnabledTools -is [string]) {
			$GLOBAL:EnabledTools = $_args.EnabledTools | ConvertFrom-Json
		}
		else {
			$GLOBAL:EnabledTools = $_args.EnabledTools
		}
	}

	if ($null -ne $_args.Tools) {

		# Tools may be JSON or native objects
		if ($_args.Tools -is [string]) {
			$GLOBAL:Tools = $_args.Tools | ConvertFrom-Json
		}
		else {
			$GLOBAL:Tools = $_args.Tools
		}
	}

	if ($null -ne $_args.EnableLogging) {
		$GLOBAL:EnableLogging = $_args.EnableLogging
	}
}

# ----------------------------------
# Build System Instructions
# ----------------------------------
function Build-SystemInstructions {
	# If no tools registered, return a minimal system prompt
	if (-not $GLOBAL:Tools -or $GLOBAL:Tools.Count -eq 0) {
		return @"
You are an agent that must respond ONLY with a single JSON object.

Since no tools are available, the ONLY valid action is "finish".
Any other action is invalid.

You MUST respond with:

{
  "action": "finish",
  "content": {
    "result": "<your final output>"
  }
}

Rules:
- Respond ONLY with valid JSON.
- Never include commentary outside the JSON.
"@
	}

	# Otherwise build full tool manifest
	$toolText = ""

	foreach ($tool in $GLOBAL:Tools) {
		$manifest = $tool.Manifest
		if (-not $manifest) { $manifest = "(No manifest provided)" }

			$toolText += @"

Tool: $($tool.Name)

$manifest

"@
	}

	return @"
You are an agent that uses tools by returning JSON:

{
  "action": "<tool_name>",
  "content": { ... }
}

Available tools:
$toolText

Rules:
- Respond ONLY with a single JSON object.
- Never include commentary outside the JSON.
- Use the correct content schema for each tool.

STATE Block Rules:
- The STATE block contains authoritative results from previous tool calls.
- The STATE block is NOT conversation history. It is NOT internal context.
- The STATE block is external data provided by the host system.
- You MUST use the STATE block to determine whether the Goal is already satisfied.
- If the STATE block contains the required information, you MUST NOT call any tools.
- When the STATE block satisfies the Goal, you MUST call the "finish" tool immediately.

SCRATCHPAD Rules:
- The scratchpad is the agent's private memory.
- It is NOT part of STATE.
- It is NOT visible to the user.
- Use scratchpad_write to store intermediate reasoning.
- Use scratchpad_read to retrieve your own notes.
- Do NOT store final results in the scratchpad.
- Final results must be returned via the "finish" tool.

"@
}

$GLOBAL:SystemInstructions = Build-SystemInstructions

# ----------------------------
# Utility: truncate context to last N chars (preserve recent entries)
# ----------------------------
function Truncate-Context {
	param([string[]]$ContextLines, [int]$MaxChars)

	$joined = ($ContextLines -join "`n")
	if ($joined.Length -le $MaxChars) { return $ContextLines }
	# keep last portion
	$tail = $joined.Substring($joined.Length - $MaxChars)
	# split into lines and return
	return $tail -split "`n"
}

function Extract-AllJsonBlocks {
    param([string]$text)

	$blocks = @()
	$start = -1
	$depth = 0

	for ($i = 0; $i -lt $text.Length; $i++) {
		if ($text[$i] -eq '{') {
			if ($depth -eq 0) { $start = $i }
			$depth++
		}
		elseif ($text[$i] -eq '}') {
			$depth--
			if ($depth -eq 0 -and $start -ge 0) {
				$blocks += $text.Substring($start, $i - $start + 1)
				$start = -1
            	}
		}
	}

	log-info "JSON blocks count: $($blocks.Count)"

    return $blocks
}

# ----------------------------------
# Handle AI Action
# ----------------------------------
function Handle-AIAction {
	param($actionObj)

	$action  = $actionObj.action
	$content = $actionObj.content

	log-info "Handle-AIAction; Action : $action"

	# Find tool by name	
	$tool = $GLOBAL:Tools | Where-Object { $_.Name -eq $action } | Select-Object -First 1

	# UNKNOWN TOOL
	if (-not $tool) {
		$GLOBAL:State["agent_error"] = "FAIL: UNKNOWN TOOL '$action'"

		log-info $GLOBAL:State["agent_error"]

		return $false
	}

	if ($action -eq "state_store") {

		if ($content -and $content.key -and $content.value) {

			# Store arbitrary key/value pair
			$GLOBAL:State[$content.key] = $content.value

			# Standard tool success markers
			$GLOBAL:State["state_store"] = "OK"
			$GLOBAL:State["state_store_error"] = ""

			log-info "state_store: stored key '$($content.key)'"			
		}
		else {
			$GLOBAL:State["state_store_error"] = "FAIL: key and value must not be empty"
			log-info $GLOBAL:State["state_store_error"]
		}

		return $false
	}

	# TOOL HAS NO HANDLER
	if (-not $tool.Handler) {
		$GLOBAL:State["$action`_error"] = "FAIL: Tool '$action' has no handler"

		log-info $GLOBAL:State["$action`_error"]

		return $false
	}

	try {
		log-info "Handler Source Code: $($tool.Handler)"

		$handlerBlock = [ScriptBlock]::Create($tool.Handler)
		$result = & $handlerBlock $content
	
		$resultString = $result | Out-String

		if($action -eq "finish"){
			$GLOBAL:Result = $resultString
		}

		if ($resultString -match '^FAIL') {
			$GLOBAL:State["$action`_error"] = $resultString
		} else {
			$GLOBAL:State["$action"] = $resultString
			$GLOBAL:State["$action`_error"] = ""
		}
	}
	catch {

		$msg = $_.Exception.Message
		$GLOBAL:State["$action`_error"] = "FAIL: $msg"
		log-error $GLOBAL:State["$action`_error"]
	}

	return ($action -eq "finish")
}

# ----------------------------------
# Convert PSCustomObject ? Hashtable (metadata-free)
# ----------------------------------
function ConvertTo-PureHashtable {
    param([Parameter(Mandatory=$true)] $obj)

    $ht = @{}

    foreach ($prop in $obj.PSObject.Properties) {
        $name  = $prop.Name
        $value = $prop.Value

        # Skip metadata
        if ($name -in @('PSObject','BaseObject','TypeNames','Members','ImmediateBaseObject')) {
            continue
        }
        if ($name -match '^__') {
            continue
        }

        # Recursively convert nested objects
        if ($value -is [System.Management.Automation.PSCustomObject]) {
            $ht[$name] = ConvertTo-PureHashtable $value
        }
        elseif ($value -is [System.Collections.IEnumerable] -and -not ($value -is [string])) {
            $arr = @()
            foreach ($item in $value) {
                if ($item -is [System.Management.Automation.PSCustomObject]) {
                    $arr += ConvertTo-PureHashtable $item
                }
                else {
                    $arr += $item
                }
            }
            $ht[$name] = $arr
        }
        else {
            $ht[$name] = $value
        }
    }

    return $ht
}

# ----------------------------------
# Validate a tool call using pure hashtable
# ----------------------------------
function Validate-ToolCall {
    param(
        [Parameter(Mandatory=$true)]
        [string]$JsonText
    )

    # Parse JSON into PSCustomObject
    try {
        $obj = $JsonText | ConvertFrom-Json -ErrorAction Stop
    }
    catch {
        $GLOBAL:State["agent_error"] = "FAIL: invalid JSON: $($_.Exception.Message)"
        return $false
    }

    # Convert to pure hashtable (no metadata)
    $ht = ConvertTo-PureHashtable $obj

    if (-not ($ht -is [hashtable])) {
        $GLOBAL:State["agent_error"] = "FAIL: JSON must be an object"
        return $false
    }

    # Basic structure
    if (-not $ht.ContainsKey('action')) {
        $GLOBAL:State["agent_error"] = "FAIL: Missing 'action'"
        return $false
    }
    if (-not $ht.ContainsKey('content')) {
        $GLOBAL:State["agent_error"] = "FAIL: Missing 'content'"
        return $false
    }

    $action  = $ht['action']
    $content = $ht['content']

    # Tool existence
    $tool = $GLOBAL:Tools | Where-Object { $_.Name -eq $action } | Select-Object -First 1
    if (-not $tool) {
        $GLOBAL:State["agent_error"] = "FAIL: UNKNOWN TOOL '$action'"
        return $false
    }

    if (-not ($content -is [hashtable])) {
        $GLOBAL:State["agent_error"] = "FAIL: content must be an object"
        return $false
    }

    # Schema lookup
    $schema     = Get-ToolSchema -ToolName $action
    $required   = $schema.RequiredFields
    $optional   = $schema.OptionalFields
    $allowExtra = $schema.AllowExtra

    $keys = $content.Keys

    # Required fields
    foreach ($field in $required) {
        if (-not ($keys -contains $field)) {
            $GLOBAL:State["agent_error"] = "FAIL: Tool '$action' missing required field '$field'"
            return $false
        }
    }

    # Extra fields
    if (-not $allowExtra) {
        $allowed = $required + $optional
        foreach ($field in $keys) {
            if (-not ($allowed -contains $field)) {
                $GLOBAL:State["agent_error"] = "FAIL: Tool '$action' has unexpected field '$field'"
                return $false
            }
        }
    }

    # Delegate strictness
    if ($action -eq 'delegate') {
        if ($keys.Count -ne 1 -or -not ($keys -contains 'AgentGoal')) {
            $GLOBAL:State["agent_error"] = "FAIL: delegate tool must contain ONLY 'AgentGoal'"
            return $false
        }
    }

    # web_search strictness
    if ($action -eq 'web_search') {
        if ($keys.Count -ne 1 -or -not ($keys -contains 'query')) {
            $GLOBAL:State["agent_error"] = "FAIL: web_search tool must contain ONLY 'query'"
            return $false
        }
    }

    return $true
}


# ----------------------------
# Main Run-Agent loop
# ----------------------------

$iter = 0

# only iterate if tools were provided
if ($GLOBAL:Tools.Count -eq 0) {
    $iter = $GLOBAL:MaxIterations
}

# Build static AI params

if($GLOBAL:AIProvider -and $GLOBAL:APIKey){

	# cloud model

	if ($GLOBAL:AIHelper -eq $null) {

		$GLOBAL:AIHelper = [AIHelper]::new()
		$GLOBAL:AIHelper.Provider      = $GLOBAL:AIProvider
		$GLOBAL:AIHelper.Model         = $GLOBAL:AIModel
		$GLOBAL:AIHelper.ApiKey        = $GLOBAL:APIKey
		$GLOBAL:AIHelper.SystemPrompt  = $GLOBAL:SystemInstructions
	}
}
else{
	# local model
	$aiParams = @{
		Model  = $GLOBAL:AIModel
		System = $GLOBAL:SystemInstructions
	}
}

while ($iter -lt [int]$GLOBAL:MaxIterations) {
	$iter++

	# Yield UI
	[System.Windows.Forms.Application]::DoEvents()

	# ----------------------------------------
	# Add agent metadata into STATE
	# ----------------------------------------

	# Track iteration count
	$GLOBAL:State["iteration"] = $iter

	# Track number of available tools
	$GLOBAL:State["tool_count"] = $GLOBAL:Tools.Count

	# Track last action (if any)
	if ($actionObj -ne $null -and $actionObj.action) {
	    $GLOBAL:State["last_action"] = $actionObj.action
	}

	if ($GLOBAL:State.Count -eq 0) {
		$stateBlock = "<empty>"
	} else {
		$stateBlock = ($GLOBAL:State.GetEnumerator() | ForEach-Object {
			$key = $_.Key
			$value = $_.Value.ToString().Replace("`n", "`n  ")
			"  $($key): $($value)"
		}) -join "`n"
	}


	$scratchBlock = ($GLOBAL:ScratchPad.GetEnumerator() | ForEach-Object {
	    "  $($_.Key): $($_.Value)"
	}) -join "`n"

	$agentPrompt = @"
You are an agent running inside AdvantageBuilder.

Goal:
$($GLOBAL:AgentGoal)

Use any available tools, if needed.
Once you get the results in a state, use it, no need to re-run the tools.

SCRATCHPAD:
This is your private memory. Use it for planning.
$scratchBlock

STATE:
The following is the authoritative state of the world.
This STATE block contains results from previous tool calls.
You MUST treat STATE as the single source of truth.
You MUST NOT re-run any tool if STATE already contains the needed information.

$stateBlock

Respond ONLY with valid JSON.
"@

	log-info "Loop iteration: $iter"
	log-info "Agent Prompt: $agentPrompt"
	log-info "System Instructions: $GLOBAL:SystemInstructions"

	# Call AI helper (returns plain string)

	if ($GLOBAL:AIHelper -eq $null){

		$aiParams.Prompt = $agentPrompt
		$response = callMacro $false 'AI Ollama Helper' 'Macro Main Helpers\AI' $aiParams

	}
	else{
		$response = $GLOBAL:AIHelper.Send($agentPrompt)
	}
	
	log-info $response

	$aiText = if ($null -eq $response) { "" } else { $response.ToString() }

	if ([string]::IsNullOrWhiteSpace($aiText)) {
		$GLOBAL:State["agent_error"] = "FAIL: AI returned empty response"
		Start-Sleep -Seconds $GLOBAL:IterationDelay
		continue
	}

	log-info "parsing..."

	$finished = $false

	# Parse first JSON block
	$actionObj = $null
	try {

		$jsonBlocks = Extract-AllJsonBlocks $aiText

		foreach ($json in $jsonBlocks) {

			log-info "Start ConvertFrom-Json : $json"

			if (-not (Validate-ToolCall -JsonText $json)) {
				log-error "Tool call validation failed for JSON: $json"

				break
			}

			$actionObj = $json | ConvertFrom-Json -ErrorAction Stop

			log-info "End ConvertFrom-Json; Action: $($actionObj.action)"

			if (Handle-AIAction -actionObj $actionObj) {
				$finished = $true
				break   # stop processing remaining JSON blocks
			}

			log-info "Finished: $finished"

			if ($finished) { break }
		}

	}
	catch {

		log-error "AI unparsable on iter $($iter): $($aiText)"
		log-error $_.Exception.Message

		# Authoritative state update
		$GLOBAL:State["agent_error"] = "FAIL: AI unparsable JSON"

		Start-Sleep -Seconds $GLOBAL:IterationDelay
		continue
	}

	if ($finished) {
		break   # stop the OUTER agent loop
	}
	
	Start-Sleep -Seconds $GLOBAL:IterationDelay
}

if ($iter -ge $GLOBAL:MaxIterations) {
	alert "Agent reached max iterations ($($GLOBAL:MaxIterations)) without finishing."

      # Authoritative state update
	$GLOBAL:State["agent_error"] = "FAIL: Agent timed out after $iter iterations"
}

$GLOBAL:Result
