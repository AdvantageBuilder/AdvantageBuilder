$engine = "PowerShell"

if ($null -ne $_args -and $_args -is [string]) {
    switch ($_args.ToLower()) {
        "powershell"     { $engine = "PowerShell" }
        "ps"             { $engine = "PowerShell" }
        "javascriptv8"   { $engine = "JavaScriptV8" }
        "jsv8"           { $engine = "JavaScriptV8" }
        "v8"             { $engine = "JavaScriptV8" }
        "jscript"        { $engine = "JScript" }
        "js"             { $engine = "JScript" }
    }
}

$cboEngine.SelectedItem = $engine

$winformsHelper.FinaliseDialogForm($mainForm)

$snippet = ""

if ($mainForm.ShowDialog() -eq [System.Windows.Forms.DialogResult]::OK) {

    $engine          = $cboEngine.SelectedItem
    $model           = $cboModel.Text
    
    if (-not $model) { $model = "llama3.1" }

    $images = @($lstImages.Items)

    if ($engine -eq "PowerShell") {  

        # Determine whether options were explicitly provided
        $optionsJsonOverride = -not [string]::IsNullOrWhiteSpace($txtOptionsJson.Text)
        $optionsFieldsProvided = -not [string]::IsNullOrWhiteSpace($txtTemp.Text) -or `
                                 -not [string]::IsNullOrWhiteSpace($txtTopP.Text) -or `
                                 -not [string]::IsNullOrWhiteSpace($txtMaxTokens.Text) -or `
                                 -not [string]::IsNullOrWhiteSpace($txtKeepAlive.Text) -or `
                                 $chkRaw.Checked -or `
                                 ($cboFormat.Text -ne "json")

        $emitOptions = $optionsJsonOverride -or $optionsFieldsProvided

        # Build options object (only used to produce JSON if needed)
        $optionsObj = @{
            temperature = if ([string]::IsNullOrWhiteSpace($txtTemp.Text)) { $null } else { [double]$txtTemp.Text }
            top_p       = if ([string]::IsNullOrWhiteSpace($txtTopP.Text))   { $null } else { [double]$txtTopP.Text }
            max_tokens  = if ([string]::IsNullOrWhiteSpace($txtMaxTokens.Text)) { $null } else { [int]$txtMaxTokens.Text }
            format      = $cboFormat.Text
            keep_alive  = $txtKeepAlive.Text
            raw         = $chkRaw.Checked
        }

        if ($optionsJsonOverride) {
            try {
                $optionsObj = $txtOptionsJson.Text | ConvertFrom-Json
            } catch {
                # invalid override JSON — ignore and keep constructed optionsObj
            }
        }

        # Build prompt payload values
        $promptText = $txtPrompt.Text
        $systemText = if ($chkSystemPrompt.Checked -and -not [string]::IsNullOrWhiteSpace($txtSystemPrompt.Text)) { $txtSystemPrompt.Text } else { $null }

        $optionsJson  = $optionsObj    | ConvertTo-Json -Depth 10

        # Build snippet 
        $sb = New-Object System.Text.StringBuilder

        # top-level header (no indent)
        $sb.AppendLine('$aiParams = @{') | Out-Null
        $sb.AppendLine('    Model = "' + $model + '"') | Out-Null
        $sb.AppendLine('}') | Out-Null
        $sb.AppendLine('') | Out-Null
            
        # Prompt here-string (starts at column 0)
        $sb.AppendLine('$aiParams.Prompt = @"') | Out-Null
        if ($null -ne $promptText) {
            foreach ($line in ($promptText -split "`r?`n")) { $sb.AppendLine($line) | Out-Null }
        }
        $sb.AppendLine('"@') | Out-Null
        $sb.AppendLine('') | Out-Null

        if ($null -ne $systemText) {
            $sb.AppendLine('$aiParams.System = @"') | Out-Null
            foreach ($line in ($systemText -split "`r?`n")) { $sb.AppendLine($line) | Out-Null }
            $sb.AppendLine('"@') | Out-Null
            $sb.AppendLine('') | Out-Null
        }

        if ($images){            
            $sb.AppendLine('$aiParams.Images = @(') | Out-Null

            foreach ($item in $images) { $sb.AppendLine("    '$item'") | Out-Null }
            $sb.AppendLine(')') | Out-Null
            $sb.AppendLine('') | Out-Null
        }
   
        if ($emitOptions) {
            $sb.AppendLine('$optionsJson = @"') | Out-Null
            foreach ($line in ($optionsJson -split "`r?`n")) { $sb.AppendLine($line) | Out-Null }
            $sb.AppendLine('"@') | Out-Null
            $sb.AppendLine('$aiParams.Options = ($optionsJson | ConvertFrom-Json)') | Out-Null
            $sb.AppendLine('') | Out-Null
        }

        $sb.AppendLine('$response = callMacro $false ''AI Ollama Helper'' ''Macro Main Helpers\AI'' $aiParams') | Out-Null

        $snippet = $sb.ToString()
        
    }
    elseif ($engine -eq "JavaScriptV8") {

        # Determine whether options were explicitly provided
        $optionsJsonOverride = -not [string]::IsNullOrWhiteSpace($txtOptionsJson.Text)
        $optionsFieldsProvided = -not [string]::IsNullOrWhiteSpace($txtTemp.Text) -or `
                                 -not [string]::IsNullOrWhiteSpace($txtTopP.Text) -or `
                                 -not [string]::IsNullOrWhiteSpace($txtMaxTokens.Text) -or `
                                 -not [string]::IsNullOrWhiteSpace($txtKeepAlive.Text) -or `
                                 $chkRaw.Checked -or `
                                 ($cboFormat.Text -ne "json")

        $emitOptions = $optionsJsonOverride -or $optionsFieldsProvided

        # Build options object (only used to produce JSON if needed)
        $optionsObj = @{
            temperature = if ([string]::IsNullOrWhiteSpace($txtTemp.Text)) { $null } else { [double]$txtTemp.Text }
            top_p       = if ([string]::IsNullOrWhiteSpace($txtTopP.Text))   { $null } else { [double]$txtTopP.Text }
            max_tokens  = if ([string]::IsNullOrWhiteSpace($txtMaxTokens.Text)) { $null } else { [int]$txtMaxTokens.Text }
            format      = $cboFormat.Text
            keep_alive  = $txtKeepAlive.Text
            raw         = $chkRaw.Checked
        }

        if ($optionsJsonOverride) {
            try {
                $optionsObj = $txtOptionsJson.Text | ConvertFrom-Json
            } catch {
                # invalid override JSON — ignore and keep constructed optionsObj
            }
        }

        # Build prompt payload values
        $promptText = $txtPrompt.Text
        $systemText = if ($chkSystemPrompt.Checked -and -not [string]::IsNullOrWhiteSpace($txtSystemPrompt.Text)) { $txtSystemPrompt.Text } else { $null }

        $optionsJson = $optionsObj | ConvertTo-Json -Depth 10

        # Build snippet
        $sb = New-Object System.Text.StringBuilder

        # top-level header (no indent)
        $sb.AppendLine('let aiParams = {') | Out-Null
        $sb.AppendLine('    Model: "' + $model + '"') | Out-Null
        $sb.AppendLine('};') | Out-Null
        $sb.AppendLine('') | Out-Null

        # Prompt template literal (starts at column 0)
        $sb.AppendLine('aiParams.Prompt = `') | Out-Null
        if ($null -ne $promptText) {
            foreach ($line in ($promptText -split "`r?`n")) { $sb.AppendLine($line) | Out-Null }
        }
        $sb.AppendLine('`;') | Out-Null
        $sb.AppendLine('') | Out-Null

        if ($null -ne $systemText) {
            $sb.AppendLine('aiParams.System = `') | Out-Null
            foreach ($line in ($systemText -split "`r?`n")) { $sb.AppendLine($line) | Out-Null }
            $sb.AppendLine('`;') | Out-Null
            $sb.AppendLine('') | Out-Null
        }

        if ($images) {

		$sb.AppendLine('aiParams.Images = [];') | Out-Null

		foreach ($item in $images) {

			$escaped = $item.Replace("'", "\\'")

			$escaped = $escaped.Replace("\", "\\")

			$sb.AppendLine("aiParams.Images.push('$escaped');") | Out-Null
		}

		$sb.AppendLine('') | Out-Null
        }

        if ($emitOptions) {
            $sb.AppendLine('let optionsJson = `') | Out-Null
            foreach ($line in ($optionsJson -split "`r?`n")) { $sb.AppendLine($line) | Out-Null }
            $sb.AppendLine('`;') | Out-Null
            $sb.AppendLine('aiParams.Options = JSON.parse(optionsJson);') | Out-Null
            $sb.AppendLine('') | Out-Null
        }

        $sb.AppendLine('let response = callMacro(false, "AI Ollama Helper", "Macro Main Helpers\\AI", aiParams);') | Out-Null

        $snippet = $sb.ToString()
    }
    elseif ($engine -eq "JScript") {

        # Determine whether options were explicitly provided
        $optionsJsonOverride = -not [string]::IsNullOrWhiteSpace($txtOptionsJson.Text)
        $optionsFieldsProvided = -not [string]::IsNullOrWhiteSpace($txtTemp.Text) -or `
                                 -not [string]::IsNullOrWhiteSpace($txtTopP.Text) -or `
                                 -not [string]::IsNullOrWhiteSpace($txtMaxTokens.Text) -or `
                                 -not [string]::IsNullOrWhiteSpace($txtKeepAlive.Text) -or `
                                 $chkRaw.Checked -or `
                                 ($cboFormat.Text -ne "json")

        $emitOptions = $optionsJsonOverride -or $optionsFieldsProvided

        # Build options object (only used to produce JSON if needed)
        $optionsObj = @{
            temperature = if ([string]::IsNullOrWhiteSpace($txtTemp.Text)) { $null } else { [double]$txtTemp.Text }
            top_p       = if ([string]::IsNullOrWhiteSpace($txtTopP.Text))   { $null } else { [double]$txtTopP.Text }
            max_tokens  = if ([string]::IsNullOrWhiteSpace($txtMaxTokens.Text)) { $null } else { [int]$txtMaxTokens.Text }
            format      = $cboFormat.Text
            keep_alive  = $txtKeepAlive.Text
            raw         = $chkRaw.Checked
        }

        if ($optionsJsonOverride) {
            try {
                $optionsObj = $txtOptionsJson.Text | ConvertFrom-Json
            } catch {
                # invalid override JSON — ignore and keep constructed optionsObj
            }
        }

        # Build prompt payload values
        $promptText = $txtPrompt.Text
        $systemText = if ($chkSystemPrompt.Checked -and -not [string]::IsNullOrWhiteSpace($txtSystemPrompt.Text)) { $txtSystemPrompt.Text } else { $null }

        $optionsJson = $optionsObj | ConvertTo-Json -Depth 10

        # Build snippet
        $sb = New-Object System.Text.StringBuilder

        # top-level header (no indent)
        $sb.AppendLine('var aiParams = new Object();') | Out-Null
        $sb.AppendLine('aiParams.Model = "' + $model + '";') | Out-Null
        $sb.AppendLine('') | Out-Null

        # Prompt multiline builder
        $sb.AppendLine('var promptText = "";') | Out-Null
        if ($null -ne $promptText) {
            foreach ($line in ($promptText -split "`r?`n")) {
                $escaped = $line.Replace('"', '""')
                $sb.AppendLine('promptText += "' + $escaped + '\n";') | Out-Null
            }
        }
        $sb.AppendLine('aiParams.Prompt = promptText;') | Out-Null
        $sb.AppendLine('') | Out-Null

        # System multiline builder
        if ($null -ne $systemText) {
            $sb.AppendLine('var systemText = "";') | Out-Null
            foreach ($line in ($systemText -split "`r?`n")) {
                $escaped = $line.Replace('"', '""')
                $sb.AppendLine('systemText += "' + $escaped + '\n";') | Out-Null
            }
            $sb.AppendLine('aiParams.System = systemText;') | Out-Null
            $sb.AppendLine('') | Out-Null
        }

        if ($images) {

		$sb.AppendLine('aiParams.Images = [];') | Out-Null

		foreach ($item in $images) {

			$escaped = $item.Replace("'", "\\'")

			$escaped = $escaped.Replace("\", "\\")

			$sb.AppendLine("aiParams.Images.push('$escaped');") | Out-Null
		}

		$sb.AppendLine('') | Out-Null
        }

        if ($emitOptions) {
            $sb.AppendLine('var optionsJson = "') | Out-Null
            foreach ($line in ($optionsJson -split "`r?`n")) {
                $escaped = $line.Replace('"', '""')
                $sb.AppendLine($escaped) | Out-Null
            }
            $sb.AppendLine('";') | Out-Null
            $sb.AppendLine('aiParams.Options = JSON.parse(optionsJson);') | Out-Null
            $sb.AppendLine('') | Out-Null
        }

        $sb.AppendLine('var response = callMacro(false, "AI Ollama Helper", "Macro Main Helpers\\AI", aiParams);') | Out-Null

        $snippet = $sb.ToString()
    }

    $snippet
}

$winformsHelper.CleanUpForm($mainForm)
